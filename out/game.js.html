/**
 * Creates and starts a new game.
 *
 * @returns {Promise<void>}
 */
async function startNewGame() {
    const levelModul = await import("./levels/level1.js");
    const freshLevel = levelModul.createLevel1();

    keyboard = new Keyboard();
    world = new World(canvas, keyboard, freshLevel);
}

/**
 * Toggles fullscreen mode for the game canvas.
 *
 * @returns {void}
 */
window.toggleFullscreen = function() {
    let canvas = document.getElementById('canvas');

    if (!document.fullscreenElement) {
        canvas.requestFullscreen();
    } else {
        document.exitFullscreen();
    }
}

window.addEventListener('keydown', (e) => {
    if (e.keyCode == 39){
        keyboard.RIGHT = true;
    }
    if (e.keyCode == 37){
        keyboard.LEFT = true;
    }
    if (e.keyCode == 38){
        keyboard.UP = true;
    }
    if (e.keyCode == 40){
        keyboard.DOWN = true;
    }
    if (e.keyCode == 32){
        keyboard.SPACE = true;

    if (e.keyCode == 67){
        keyboard.C = true;
    }
});

/**
 * Binds all mobile control buttons.
 *
 * @returns {void}
 */
function bindMobilControls() {
    const mobileLeft = document.getElementById('mobileLeft');
    const mobileRight = document.getElementById('mobileRight');
    const mobileJump = document.getElementById('mobileJump');
    const mobileThrow = document.getElementById('mobileThrow');

    mobileLeft.addEventListener('pointerdown', () => {
        keyboard.LEFT = true;
